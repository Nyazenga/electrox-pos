# HTTP server - redirect to HTTPS
server {
    listen 80;
    server_name nedcom.co.zw www.nedcom.co.zw;
    return 301 https://$server_name$request_uri;
}

# HTTPS server
server {
    listen 443 ssl http2;
    server_name nedcom.co.zw www.nedcom.co.zw;
    root /var/www/electro-pos;
    index index.php index.html;

    # SSL configuration
    ssl_certificate /etc/ssl/certs/nedcom.co.zw.crt;
    ssl_certificate_key /etc/ssl/private/nedcom.co.zw.key;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;
    ssl_prefer_server_ciphers on;

    access_log /var/log/nginx/nedcom_access.log;
    error_log /var/log/nginx/nedcom_error.log;

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        fastcgi_pass unix:/var/run/php/php8.3-fpm.sock;
        fastcgi_index index.php;
        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        include fastcgi_params;
    }

    location ~ /\.ht {
        deny all;
    }
}
